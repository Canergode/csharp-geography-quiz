using System;

namespace GeographyQuiz
{
    class Program
    {
        static void Main(string[] args)
        {
            StartQuiz();
            Console.ReadLine();
        }

        private static void StartQuiz()
        {
            // Questions and answer arrays
            string[] questions = {
                "Which city has plate number 33?",
                "Which district is famous for bananas?",
                "Which region is Mersin in?"
            };

            string[] answers = new string[3];

            // Ask questions and collect answers
            for (int i = 0; i < 3; i++)
            {
                Console.WriteLine(questions[i]);
                answers[i] = Console.ReadLine();
            }

            int correct = 0;
            int wrong   = 0;

            // Check answers
            if (answers[0] == "Mersin")  correct++; else wrong++;
            if (answers[1] == "Anamur")  correct++; else wrong++;
            if (answers[2] == "Mediterranean") correct++; else wrong++;

            Console.WriteLine("Correct: " + correct);
            Console.WriteLine("Wrong:   " + wrong);
        }
    }
}
